# launch or start k9s with context
k9s --context CONTEXT_NAME

# launch in a specific namespace
k9s -n NAMESPACE

# view all namespaces
0

# jump to a resource (pods, svc, deploy, etc)
: then RESOURCE

# filter current view
/ then SEARCH_TERM

# describe selected resource
d

# view logs of selected resource
l

# open a shell in selected pod
s

# delete selected resource
ctrl + d

# go back or clear filter
esc

# quit
:q
