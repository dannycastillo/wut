# repeat last command
!!

# repeat last command with sudo
sudo !!

# show the exit code of the last command
echo $?

# send stdout and stderr to a file
COMMAND > FILE 2>&1

# write output to a file and still see it
COMMAND | tee FILE

# run a command once per line of a file
cat FILE | xargs -I {} COMMAND {}

# run a command every n seconds
while true; do clear; COMMAND; sleep N; done

# count the unique lines of a file, most common first
sort FILE | uniq -c | sort -rn
