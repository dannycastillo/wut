# Navigate to end or start of line
cmd + right or cmd + left

# Navigate to next word or last word (Navigate back or forward words)
option + right or option + left

# Clear line
ctl + u

# Clear from cursor to end of line
ctl + k

# Clear screen
ctl + l

# Search command history
ctl + r

# Repeat last command
!!

# Repeat last command with sudo
sudo !!

# check system disk use
df -h /System/Volumes/Data
