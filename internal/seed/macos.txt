# copy a file's contents to the clipboard
pbcopy < FILE

# paste the clipboard into a file
pbpaste > FILE

# open a file in a specific app
open -a NAME FILE

# keep the mac awake while a command runs
caffeinate -i COMMAND

# keep the mac and display awake until ctrl-c
caffeinate -d

# search file names with spotlight
mdfind -name PATTERN

# show hidden files in finder
defaults write com.apple.finder AppleShowAllFiles -bool true && killall Finder
