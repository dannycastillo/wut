# run a container interactively and remove it on exit
docker run -it --rm IMAGE bash

# run a container in the background with a port published (local:container)
docker run -d --name NAME -p PORT:PORT IMAGE

# open a shell in a running container
docker exec -it CONTAINER bash

# follow container logs
docker logs -f CONTAINER

# build an image from the current directory
docker build -t IMAGE .

# tag an image for a registry and push it
docker tag IMAGE HOST/IMAGE:TAG && docker push HOST/IMAGE:TAG

# copy a file out of a container
docker cp CONTAINER:SRC DST

# delete all buildkit cache without asking
docker builder prune -af

# delete all unused images without asking
docker image prune -af

# delete all unused containers, networks, images and cache
docker system prune -af

# rebuild images then start compose services in the background
docker compose up -d --build

# follow logs from every compose service
docker compose logs -f
