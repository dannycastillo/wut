# initialize and upgrade providers to the newest allowed versions
terraform init -upgrade

# save a plan, then apply exactly that plan
terraform plan -out=tfplan && terraform apply tfplan

# apply without confirmation prompt
terraform apply -auto-approve

# apply to one resource only
terraform apply -target=RESOURCE

# destroy and recreate one resource
terraform apply -replace=RESOURCE

# list resources in state
terraform state list

# show one resource from state
terraform state show RESOURCE

# forget a resource without destroying it
terraform state rm RESOURCE

# bring an existing resource under management
terraform import RESOURCE ID

# select a workspace
terraform workspace select WORKSPACE

# opentofu is a drop-in fork, so every terraform snippet here runs as tofu
tofu init
