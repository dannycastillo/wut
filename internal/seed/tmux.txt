# start a detached session running a command
tmux new -d -s SESSION COMMAND

# attach to a session, creating it if it does not exist
tmux new -A -s SESSION

# detach from the current session (prefix d)
tmux detach

# rename the current session (prefix $)
tmux rename-session NAME

# kill all sessions
tmux kill-server

# split the pane side by side (prefix %)
tmux split-window -h

# split the pane top and bottom (prefix ")
tmux split-window -v

# move to the pane in a direction, one of -l -r -u -d (prefix arrow)
tmux select-pane -L

# zoom the current pane to full size, or back (prefix z)
tmux resize-pane -Z

# type a command into another session
tmux send-keys -t SESSION "COMMAND" Enter

# save the current pane's scrollback to a file
tmux capture-pane -p -S - > FILE

# reload the config
tmux source-file ~/.tmux.conf
